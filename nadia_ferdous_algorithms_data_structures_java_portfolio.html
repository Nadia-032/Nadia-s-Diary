DijkstraShortestPath.java 
public int[] shortestPath(int V, List<List<Node>> adj, int src) {
    int[] dist = new int[V];
    Arrays.fill(dist, Integer.MAX_VALUE); dist[src] = 0;
    PriorityQueue<Node> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a.weight));
    pq.offer(new Node(src, 0));
    while (!pq.isEmpty()) {
        Node curr = pq.poll();
        for (Node edge : adj.get(curr.vertex)) {
            if (dist[curr.vertex] + edge.weight < dist[edge.vertex]) {
                dist[edge.vertex] = dist[curr.vertex] + edge.weight;
                pq.offer(new Node(edge.vertex, dist[edge.vertex]));
            }
        }
    } return dist; // Time: O((V + E) log V)
}
